cmake_minimum_required(VERSION 3.16)
project(Calculator LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(calc src/main.cpp src/plugin_loader.cpp src/parser.cpp)
target_include_directories(calc PRIVATE include)

if(WIN32)
    target_compile_definitions(calc PRIVATE NOMINMAX)
else()
    target_link_libraries(calc PRIVATE ${CMAKE_DL_LIBS})
endif()

function(add_calculator_plugin target source)
    add_library(${target} SHARED ${source})
    target_include_directories(${target} PRIVATE include)
    target_compile_definitions(${target} PRIVATE CALCULATOR_PLUGIN_BUILD)
    set_target_properties(${target} PROPERTIES PREFIX "" OUTPUT_NAME "${target}")
    if(APPLE)
        set_target_properties(${target} PROPERTIES SUFFIX ".dylib")
    elseif(WIN32)
        set_target_properties(${target} PROPERTIES SUFFIX ".dll" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
    else()
        set_target_properties(${target} PROPERTIES SUFFIX ".so")
    endif()
    if(NOT WIN32)
        set_target_properties(${target} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
    endif()
endfunction()

add_calculator_plugin(funcsin plugins/sin.cpp)
add_calculator_plugin(funcdeg plugins/deg.cpp)
add_calculator_plugin(funcln plugins/ln.cpp)
add_calculator_plugin(funcexp plugins/exp.cpp)

add_dependencies(calc funcsin funcdeg funcln funcexp)
